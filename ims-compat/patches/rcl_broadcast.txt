    /**
     * Broadcast an action to all registered callbacks (Android 9 API).
     * @hide
     */
    public void broadcast(java.util.function.Consumer<T> action) {
        int N = beginBroadcast();
        try {
            for (int i = 0; i < N; i++) {
                try {
                    action.accept(getBroadcastItem(i));
                } catch (Exception e) {
                    // ignore callback failure
                }
            }
        } finally {
            finishBroadcast();
        }
    }
